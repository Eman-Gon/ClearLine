cmake_minimum_required(VERSION 3.22.1)
project(clearline_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(WHISPER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.native/whisper.cpp" CACHE PATH "Pinned whisper source")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "Run audio/scripts/fetch-whisper.sh before building. No runtime source downloads.")
endif()
execute_process(COMMAND git -C "${WHISPER_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE WHISPER_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT WHISPER_REV STREQUAL "a8d002cfd879315632a579e73f0148d06959de36")
  message(FATAL_ERROR "whisper.cpp checkout does not match the pinned commit")
endif()
execute_process(COMMAND git -C "${WHISPER_SOURCE_DIR}" status --porcelain OUTPUT_VARIABLE WHISPER_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT WHISPER_DIRTY STREQUAL "")
  message(FATAL_ERROR "whisper.cpp source has local changes; immutable source required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" "whisper-build" EXCLUDE_FROM_ALL)
add_library(clearline_whisper SHARED whisper_jni.cpp)
target_link_libraries(clearline_whisper PRIVATE whisper android log)
target_compile_options(clearline_whisper PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
# Hide all vendored native symbols so llama.cpp's distinct ggml cannot interpose.
target_link_options(clearline_whisper PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,-z,max-page-size=16384")
